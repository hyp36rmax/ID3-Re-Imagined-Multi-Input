cmake_minimum_required(VERSION 3.24)
project(Id3DeviceIdentityProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The inventory DLL requires Windows x64. Portable tests are separate.")
endif()
set(PROBE_BUILD_ID "unidentified-build" CACHE STRING "Git commit and source digest")
add_library(Id3IdentityInventory SHARED inventory.cpp)
target_compile_features(Id3IdentityInventory PRIVATE cxx_std_17)
target_compile_definitions(Id3IdentityInventory PRIVATE UNICODE _UNICODE PROBE_BUILD_ID="${PROBE_BUILD_ID}")
target_link_libraries(Id3IdentityInventory PRIVATE setupapi cfgmgr32 hid dinput8 dxguid xinput9_1_0 ole32 user32)
if(MSVC)
  target_compile_options(Id3IdentityInventory PRIVATE /W4 /EHsc)
endif()
